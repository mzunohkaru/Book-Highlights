// ItemList.test.js - 実践編
import { mount } from '@vue/test-utils'
import ItemList from '@/components/ItemList.vue'

describe('ItemList コンポーネント - 実践編', () => {
  
  // 🌟 テーブル駆動テスト（複数パターンを効率的に）
  describe('様々なデータパターンでの表示テスト', () => {
    // テストデータをまとめて定義
    const testCases = [
      {
        name: '通常データ',
        items: [{ type: 'one', title: 'タイトル1' }],
        expectedCount: 1,
        shouldShowEmpty: false
      },
      {
        name: '複数データ',
        items: [
          { type: 'one', title: 'タイトル1' },
          { type: 'two', title: 'タイトル2' },
          { type: 'three', title: 'タイトル3' }
        ],
        expectedCount: 3,
        shouldShowEmpty: false
      },
      {
        name: '空データ',
        items: [],
        expectedCount: 0,
        shouldShowEmpty: true
      }
    ]

    // 各パターンをテスト
    testCases.forEach(({ name, items, expectedCount, shouldShowEmpty }) => {
      test(`${name}の場合、正しく表示される`, () => {
        const wrapper = mount(ItemList, {
          props: { items }
        })

        if (shouldShowEmpty) {
          expect(wrapper.find('[data-testid="empty-message"]').exists()).toBe(true)
        } else {
          expect(wrapper.findAll('[data-testid^="item-"]')).toHaveLength(expectedCount)
        }
      })
    })
  })

  // 🛡️ 境界値テスト（エラーが起きやすいところ）
  describe('境界値・異常値のテスト', () => {
    test('nullが渡された時でもエラーにならない', () => {
      // 本来あり得ないけど、バグで起こる可能性がある
      expect(() => {
        mount(ItemList, {
          props: { items: null }
        })
      }).not.toThrow()
    })

    test('undefinedが渡された時でもエラーにならない', () => {
      expect(() => {
        mount(ItemList, {
          props: { items: undefined }
        })
      }).not.toThrow()
    })

    test('文字列が渡された時でもエラーにならない', () => {
      expect(() => {
        mount(ItemList, {
          props: { items: 'これは配列じゃない' }
        })
      }).not.toThrow()
    })
  })

  // 🎭 コンポーネントの振る舞いテスト
  describe('コンポーネントの動的な振る舞い', () => {
    test('propsが変更された時、表示も更新される', async () => {
      const wrapper = mount(ItemList, {
        props: { items: [] }
      })

      // 最初は空メッセージ
      expect(wrapper.find('[data-testid="empty-message"]').exists()).toBe(true)

      // propsを変更
      await wrapper.setProps({
        items: [{ type: 'one', title: 'タイトル1' }]
      })

      // リストが表示される
      expect(wrapper.find('[data-testid="item-list"]').exists()).toBe(true)
      expect(wrapper.find('[data-testid="empty-message"]').exists()).toBe(false)
    })

    test('ローディング状態が変更される', async () => {
      const wrapper = mount(ItemList, {
        props: { items: [], loading: false }
      })

      // 最初はローディングなし
      expect(wrapper.find('[data-testid="loading"]').exists()).toBe(false)

      // ローディング開始
      await wrapper.setProps({ loading: true })
      expect(wrapper.find('[data-testid="loading"]').exists()).toBe(true)

      // ローディング終了
      await wrapper.setProps({ loading: false })
      expect(wrapper.find('[data-testid="loading"]').exists()).toBe(false)
    })
  })
})
